========== infrastructure/user_data.sh ==========

#!/bin/bash

# Enhanced user data script with logging and error handling
exec > >(tee /var/log/user-data.log|logger -t user-data -s 2>/dev/console) 2>&1
echo "Starting user data script execution at $(date)"

# Set AWS region globally
export AWS_DEFAULT_REGION=ap-southeast-1
echo "✓ AWS region set to $AWS_DEFAULT_REGION"

# Wait for cloud-init to finish
echo "Waiting for system to be ready..."
cloud-init status --wait
echo "✓ Cloud-init completed"

# Wait for apt lock (max 5 minutes)
echo "Waiting for apt lock..."
for i in {1..300}; do
    if ! lsof /var/lib/apt/lists/lock > /dev/null 2>&1 && \
       ! lsof /var/lib/dpkg/lock-frontend > /dev/null 2>&1; then
        echo "✓ APT is ready"
        break
    fi
    [ $i -eq 300 ] && { echo "ERROR: APT lock timeout"; exit 1; }
    sleep 1
done

# Update system
echo "Updating system packages..."
DEBIAN_FRONTEND=noninteractive apt-get update -y || { echo "ERROR: apt-get update failed"; exit 1; }
DEBIAN_FRONTEND=noninteractive apt-get upgrade -y -o Dpkg::Options::="--force-confdef" -o Dpkg::Options::="--force-confold"

# Install required packages
echo "Installing required packages..."
DEBIAN_FRONTEND=noninteractive apt-get install -y \
    docker.io \
    awscli \
    jq \
    curl \
    unzip \
    lsof || { echo "ERROR: Package installation failed"; exit 1; }

echo "✓ All packages installed successfully"

# Start Docker
echo "Starting Docker service..."
systemctl start docker || { echo "ERROR: Failed to start Docker"; exit 1; }
systemctl enable docker
usermod -aG docker ubuntu

# Wait for Docker daemon
echo "Waiting for Docker daemon..."
for i in {1..30}; do
    if docker ps &> /dev/null; then
        echo "✓ Docker daemon is ready"
        break
    fi
    [ $i -eq 30 ] && { echo "ERROR: Docker daemon timeout"; exit 1; }
    sleep 2
done

# Wait for IAM instance profile
echo "Waiting for IAM instance profile..."
for i in {1..60}; do
    if aws sts get-caller-identity &> /dev/null; then
        ACCOUNT_ID=$(aws sts get-caller-identity --query Account --output text)
        echo "✓ IAM credentials available (Account: $ACCOUNT_ID)"
        break
    fi
    [ $i -eq 60 ] && { echo "ERROR: IAM credentials timeout"; exit 1; }
    sleep 2
done

# Function to retrieve SSM parameters
get_ssm_parameter() {
    local param_name=$1
    local max_retries=5
    local retry=0
    
    while [ $retry -lt $max_retries ]; do
        result=$(aws ssm get-parameter \
            --name "$param_name" \
            --with-decryption \
            --region ap-southeast-1 \
            --query 'Parameter.Value' \
            --output text 2>&1)
        
        if [ $? -eq 0 ] && [ -n "$result" ] && [ "$result" != "None" ]; then
            echo "$result"
            return 0
        fi
        
        retry=$((retry + 1))
        [ $retry -lt $max_retries ] && sleep 3
    done
    
    echo "ERROR: Failed to retrieve $param_name after $max_retries attempts" >&2
    return 1
}

# Create environment file from SSM parameters
echo "Fetching SSM parameters..."
> /home/ubuntu/.env

%{ for param in ssm_parameters ~}
echo "Fetching ${param}..."
PARAM_VALUE=$(get_ssm_parameter "${param}")
if [ $? -eq 0 ]; then
    echo "${replace(param, "/domrov/backend/", "")}=$PARAM_VALUE" >> /home/ubuntu/.env
else
    echo "ERROR: Failed to retrieve ${param}"
    exit 1
fi
%{ endfor ~}

# Set ownership
chown ubuntu:ubuntu /home/ubuntu/.env
chmod 600 /home/ubuntu/.env

# Verify .env file
if [ ! -s /home/ubuntu/.env ]; then
    echo "ERROR: .env file is empty"
    exit 1
fi

echo "✓ Environment file created with $(wc -l < /home/ubuntu/.env) variables"

# Pull Docker images
echo "Pulling Docker images..."
docker pull ${app_image} || { echo "ERROR: Failed to pull ${app_image}"; exit 1; }
echo "✓ App image pulled"

docker pull ${code_eval_image} || { echo "ERROR: Failed to pull ${code_eval_image}"; exit 1; }
echo "✓ Code eval image pulled"

# Create docker-compose file
cat > /home/ubuntu/docker-compose.yml << 'EOL'
version: '3.8'
services:
  domrov-code-eval:
    image: ${code_eval_image}
    container_name: domrov-code-eval
    ports:
      - "50051:50051"
    env_file:
      - .env
    restart: unless-stopped
    networks:
      - domrov-network
    
  domrov-app:
    image: ${app_image}
    container_name: domrov-app
    ports:
      - "${app_port}:${app_port}"
    env_file:
      - .env
    environment:
      - CODE_EVAL_GRPC_CLIENT_HOST=domrov-code-eval
      - CODE_EVAL_GRPC_CLIENT_PORT=50051
    restart: unless-stopped
    depends_on:
      - domrov-code-eval
    networks:
      - domrov-network
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:${app_port}/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 60s

networks:
  domrov-network:
    driver: bridge
EOL

chown ubuntu:ubuntu /home/ubuntu/docker-compose.yml

# Start containers
echo "Starting Docker containers..."
cd /home/ubuntu
docker compose up -d || /usr/local/bin/docker-compose up -d

# Wait for containers
echo "Waiting for containers to start..."
for i in {1..60}; do
    if docker ps --filter "name=domrov-app" --filter "status=running" | grep -q "domrov-app"; then
        echo "✓ Containers started successfully"
        docker ps
        break
    fi
    if [ $i -eq 60 ]; then
        echo "ERROR: Container startup timeout"
        docker ps -a
        docker logs domrov-app 2>&1 || echo "No logs"
        exit 1
    fi
    sleep 2
done

# Create systemd service
cat > /etc/systemd/system/domrov-app.service << 'EOL'
[Unit]
Description=Domrov Application
Requires=docker.service
After=docker.service

[Service]
Type=oneshot
RemainAfterExit=yes
WorkingDirectory=/home/ubuntu
ExecStart=/bin/bash -c 'cd /home/ubuntu && docker compose up -d'
ExecStop=/bin/bash -c 'cd /home/ubuntu && docker compose down'
TimeoutStartSec=0

[Install]
WantedBy=multi-user.target
EOL

systemctl enable domrov-app.service

echo "✓ User data script completed successfully at $(date)"
